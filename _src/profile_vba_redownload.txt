Function ConnectCAD() As Boolean
    Dim nm, Ls, Cs, i As Long, lo As Object
    On Error Resume Next
    For Each nm In Array("ZWCAD.Application", "ZWCAD.Application.2025", _
        "ZWCAD.Application.2024", "ZWCAD.Application.2023", "AutoCAD.Application")
        Set acad = Nothing
        Set acad = GetObject(, CStr(nm))
        If acad Is Nothing Then Set acad = CreateObject(CStr(nm))
        If Not acad Is Nothing Then Exit For
    Next
    On Error GoTo 0
    If acad Is Nothing Then MsgBox "ZWCAD ���� ����": Exit Function
    acad.Visible = True
    If acad.Documents.Count = 0 Then acad.Documents.Add
    Set doc = acad.ActiveDocument
    Set ms = doc.ModelSpace
    Ls = Array("PF_TABLE", "PF_HEAD", "PF_GRID", "PF_TEXT", "PF_GL", "PF_PIPE", "PF_SLOPE", "PF_SLOPE_T", _
               "PF_SLOPE_L", "PF_DIM", "PF_DIM_T", "PF_DIM_M", "PF_NOTE", "PF_NOTE_T", "PF_FAC", _
               "PF_GWL", "PF_WR", "PF_SR", "PF_UTIL", "PF_UTIL_L", "PF_UTIL_T", "PF_SCALE", _
               "PF_FACB", "PF_FACB_M", "PF_FACB_T")
    Cs = Array(COL_TABLE, COL_HEAD, COL_GRID, COL_TEXT, COL_GL, COL_PIPE, COL_SLOPE, COL_SLOPE_T, _
               COL_SLOPE_L, COL_DIM, COL_DIM_T, COL_DIM_M, COL_NOTE, COL_NOTE_T, COL_FAC, _
               COL_GWL, COL_WR, COL_SR, COL_UTIL, COL_UTIL_L, COL_UTIL_T, COL_SCALE, _
               COL_FACB, COL_FACB_M, COL_FACB_T)
    For i = 0 To UBound(Ls)
        Set lo = Nothing
        On Error Resume Next
        Set lo = doc.Layers.Item(CStr(Ls(i)))
        If lo Is Nothing Then Set lo = doc.Layers.Add(CStr(Ls(i)))
        lo.Color = Cs(i)
        On Error GoTo 0
    Next
    ConnectCAD = True
End Function

Function PL_Line(ByVal x1 As Double, ByVal y1 As Double, ByVal x2 As Double, ByVal y2 As Double, ByVal lay As String) As Object
    Dim o As Object

Function PL_Poly(xs() As Double, ys() As Double, ByVal n As Long, ByVal lay As String) As Object
    Dim a() As Double, i As Long, o As Object
    If n < 2 Then Exit Function
    ReDim a(2 * n - 1)
    For i = 0 To n - 1: a(2 * i) = xs(i): a(2 * i + 1) = ys(i): Next
    Set o = ms.AddLightWeightPolyline(a): o.Layer = lay
    Set PL_Poly = o
End Function

' al: 0=���ؼ����� 2=���ؼ������� 9=�߰����� 10=���߾� 11=�߰������� 13=�Ʒ����
Function PL_Text(ByVal s As String, ByVal x As Double, ByVal y As Double, ByVal hp As Double, _
                 ByVal lay As String, Optional ByVal al As Long = 10, Optional ByVal rotDeg As Double = 0) As Object
    Dim o As Object
    If Len(s) = 0 Then Exit Function
    Set o = ms.AddText(s, P3(x, y), hp * SC)
    o.Layer = lay
    On Error Resume Next: o.ScaleFactor = TEXT_WF: On Error GoTo 0
    If al <> 0 Then
        o.Alignment = al
        o.TextAlignmentPoint = P3(x, y)
    End If
    If rotDeg <> 0 Then o.Rotation = rotDeg * PI / 180#
    Set PL_Text = o
End Function

' Ÿ��(�Ǵ� ��Ÿ��)�� ����������: a0~a1 ����(��), 0��=������
Function PL_Ell(ByVal cx As Double, ByVal cy As Double, ByVal rx As Double, ByVal ry As Double, _
                ByVal a0 As Double, ByVal a1 As Double, ByVal lay As String) As Object
    Dim n As Long, i As Long, t As Double, xs() As Double, ys() As Double
    n = 36: ReDim xs(n), ys(n)
    For i = 0 To n
        t = (a0 + (a1 - a0) * i / n) * PI / 180#
        xs(i) = cx + rx * Cos(t): ys(i) = cy + ry * Sin(t)
    Next
    Set PL_Ell = PL_Poly(xs, ys, n + 1, lay)
End Function

' ä�� ȭ����: ����(xt,yt), dirS=+1�̸� ������ ������(������ ����Ŵ)
Sub PL_Tri(ByVal xt As Double, ByVal yt As Double, ByVal dirS As Double, ByVal lay As String)
    Dim o As Object, al As Double, aW As Double, xs(3) As Double, ys(3) As Double
    al = BAND_ARW * SC: aW = 0.3 * BAND_ARW * SC
    On Error Resume Next
    Set o = ms.AddSolid(P3(xt, yt), P3(xt + dirS * al, yt + aW), P3(xt + dirS * al, yt - aW), P3(xt + dirS * al, yt - aW))
    On Error GoTo 0
    If o Is Nothing Then
        xs(0) = xt + dirS * al: ys(0) = yt + aW
        xs(1) = xt: ys(1) = yt
        xs(2) = xt + dirS * al: ys(2) = yt - aW
        xs(3) = xs(0): ys(3) = ys(0)
        Set o = PL_Poly(xs, ys, 4, lay)
    Else
        o.Layer = lay
    End If
End Sub

Function StaText(ByVal d As Double) As String
    Dim n As Long, r As Double

    If r < 0.005 Then StaText = "No." & n Else StaText = "No." & n & "+" & Format(r, "0.00")
End Function

Function ParseSta(ByVal v As Variant) As Double
    Dim s As String, p As Long
    If IsNumeric(v) Then ParseSta = CDbl(v): Exit Function
    s = Replace(UCase(Trim(CStr(v))), "NO.", "")
    p = InStr(s, "+")
    If p > 0 Then
        ParseSta = Val(Left(s, p - 1)) * gChain + Val(Mid(s, p + 1))
    Else
        ParseSta = Val(s) * gChain
    End If
End Function

' ���� �� ���� (���� mm)
Function TxtW(ByVal s As String, ByVal tH As Double) As Double
    Dim i As Long, ch As Long, w As Double
    For i = 1 To Len(s)
        ch = AscW(Mid$(s, i, 1))
        If ch < 0 Or ch > 255 Then
            w = w + 1#
        ElseIf ch = 32 Then
            w = w + 0.5
        Else
            w = w + 0.72
        End If
    Next
    TxtW = w * tH * TEXT_WF
End Function

        r = r & Mid$(s, i, 1)
    Next
    SpaceOut = r
End Function

Sub FillCol(ws As Worksheet, ByVal col As Long, arr() As Double)
    Dim i As Long, a As Long, b As Long, c As Range
    Dim known() As Boolean: ReDim known(pN - 1)
    For i = 0 To pN - 1
        Set c = ws.Cells(pIdx(i), col)
        known(i) = False
        If Len(Trim(c.Value)) > 0 And IsNumeric(c.Value) Then
            If Not IsRed(c) And Val(c.Value) <> 0 Then known(i) = True: arr(i) = CDbl(c.Value)
        End If
    Next
    For i = 0 To pN - 1
        If Not known(i) Then
            a = i - 1
            Do While a >= 0
                If known(a) Then Exit Do
                a = a - 1
            Loop
            b = i + 1
            Do While b < pN
                If known(b) Then Exit Do
                b = b + 1
            Loop
            If a < 0 Or b >= pN Then
                gErr.Add "��" & pIdx(i) & " " & IIf(col = 3, "C", "D") & "��: ���� �Ұ�(�ܻ�)"
            Else
                If pD(b) = pD(a) Then
                    arr(i) = arr(a)
                Else
                    arr(i) = arr(a) + (arr(b) - arr(a)) * (pD(i) - pD(a)) / (pD(b) - pD(a))
                End If
                arr(i) = Round(arr(i), 2)
                With ws.Cells(pIdx(i), col): .Value = arr(i): .Font.Color = RGB(255, 0, 0): End With
            End If
        End If
    Next
End Sub

        Do While ng > 1
            c1 = gE(ng - 1) - gS(ng - 1) + 1: ctr1 = gSum(ng - 1) / c1
            c2 = gE(ng - 2) - gS(ng - 2) + 1: ctr2 = gSum(ng - 2) / c2
            If (ctr1 - (c1 - 1) * g / 2) - (ctr2 + (c2 - 1) * g / 2) >= g - 0.000001 Then Exit Do
            gE(ng - 2) = gE(ng - 1): gSum(ng - 2) = gSum(ng - 2) + gSum(ng - 1)
            ng = ng - 1
        Loop
    Next
    For q = 0 To ng - 1
        c1 = gE(q) - gS(q) + 1: ctr1 = gSum(q) / c1
        For i = gS(q) To gE(q)
            px(i) = ctr1 + ((i - gS(q)) - (c1 - 1) / 2) * g
        Next
    Next
End Sub

Sub DrawSlope(sd() As Double, si() As Double, sz() As Double, sx() As Long, ByVal n As Long, ByVal dS As Double)

    Dim a As Long, b As Long, ua As Long, ub As Long, ca As Double, cb As Double, ra As Double, rb As Double
    Dim L As Double, h As Double, t1 As String, t2 As String, wl As Double, xm As Double, yLab As Double
    Dim x1 As Double, x2 As Double, lastR As Double, gp As Double, xq As Double

    yM = yTblTop - R_SLP * SC / 2: r = R_SLP * SC * 0.42
    tH = TH_SLP * SC: ti = TH_SLP * 0.8 * SC
    For i = 0 To pN - 1
        If pp(i) Then hasP = True: Exit For
    Next
    ' 1) ���� ������: A�� P (P�� ������ ���� ����)
    ReDim bk(n)
    For i = 0 To n - 1
        isB = (i = 0 Or i = n - 1)
        If Not isB Then
            If hasP Then
                If sx(i) >= 0 Then isB = pp(sx(i))
            Else
                If sd(i) - sd(i - 1) < 0.005 Or sd(i + 1) - sd(i) < 0.005 Then
                    isB = True
                Else
                    s0 = (si(i) - si(i - 1)) / (sd(i) - sd(i - 1))
                    s1 = (si(i + 1) - si(i)) / (sd(i + 1) - sd(i))
                    isB = (Abs(s1 - s0) > SLOPE_CHG)
                End If
            End If
            If (sz(i) > 0) <> (sz(i - 1) > 0) Or (sz(i) > 0) <> (sz(i + 1) > 0) Then isB = True
        End If
        If isB Then bk(nb) = i: nb = nb + 1
    Next
    ' 2) ������ ������ ������ (���� ���� �ִ� 2��)
    ReDim uD(nb), ux(nb), uR(nb), uV1(nb), uV2(nb), uC(nb)
    For k = 0 To nb - 1
        i = bk(k)
        If nu > 0 Then
            If Abs(sd(i) - uD(nu - 1)) < 0.005 Then
                If si(i) > 0 Then
                    If uC(nu - 1) = 0 Then
                        uV1(nu - 1) = si(i): uC(nu - 1) = 1
                    ElseIf uC(nu - 1) = 1 And Abs(si(i) - uV1(nu - 1)) >= 0.005 Then
                        uV2(nu - 1) = si(i): uC(nu - 1) = 2
                    End If
                End If
                GoTo nextK
            End If
        End If
        uD(nu) = sd(i): ux(nu) = XD(sd(i), dS): uC(nu) = 0
        If si(i) > 0 Then uV1(nu) = si(i): uC(nu) = 1
        nu = nu + 1
nextK:
    Next
    ' 3) Ÿ�� (�̿� ���� ���ݿ� ���� �� ���, ������ ������ ��, ������ ���� ��)
    For u = 0 To nu - 1
        uR(u) = r
        If u > 0 Then
            gp = ux(u) - ux(u - 1)
            If 0.45 * gp < uR(u) Then uR(u) = 0.45 * gp
        End If
        If u < nu - 1 Then
            gp = ux(u + 1) - ux(u)
            If 0.45 * gp < uR(u) Then uR(u) = 0.45 * gp
        End If
        If uC(u) = 2 Then
            If uR(u) < 1.3 * ti Then uR(u) = 1.3 * ti
        Else
            If uR(u) < 0.7 * ti Then uR(u) = 0.7 * ti
        End If
        If uC(u) > 0 Then
            If u = 0 Then
                PL_Ell ux(u), yM, uR(u), r, -90, 90, "PF_SLOPE": xq = ux(u) + uR(u) * 0.45
            ElseIf u = nu - 1 Then
                PL_Ell ux(u), yM, uR(u), r, 90, 270, "PF_SLOPE": xq = ux(u) - uR(u) * 0.45
            Else

            Else
                PL_Text Format(uV1(u), "0.00"), xq - 0.6 * ti, yM, TH_SLP * 0.8, "PF_SLOPE_T", 10, 90
                PL_Text Format(uV2(u), "0.00"), xq + 0.6 * ti, yM, TH_SLP * 0.8, "PF_SLOPE_T", 10, 90

    Next
    ' 4) ���輱�� ����

    For k = 0 To nb - 2
        a = bk(k): b = bk(k + 1)
        If sd(b) - sd(a) < 0.005 Then GoTo nextS
        If sz(a) <= 0 Or si(a) <= 0 Or si(b) <= 0 Then GoTo nextS
        ua = 0: ub = 0
        For u = 0 To nu - 1
            If Abs(uD(u) - sd(a)) < 0.005 Then ua = u
            If Abs(uD(u) - sd(b)) < 0.005 Then ub = u
        Next
        ca = ux(ua): cb = ux(ub): ra = uR(ua): rb = uR(ub)
        L = sd(b) - sd(a): h = si(b) - si(a)
        If h > 0.0049 Then
            PL_Line ca, yM - r, cb, yM + r, "PF_SLOPE"
        ElseIf h < -0.0049 Then
            PL_Line ca, yM + r, cb, yM - r, "PF_SLOPE"
        ElseIf cb - rb > ca + ra Then
            PL_Line ca + ra, yM, cb - rb, yM, "PF_SLOPE"
        End If
        If Abs(h) < 0.005 Then
            t1 = "LEVEL": t2 = "L=" & Format(L, "0.00")
        Else
            t1 = "S=" & Format(h / L * 100#, "0.00") & "%"
            t2 = "H=" & Format(h, "0.00") & " L=" & Format(L, "0.00")
        End If
        wl = Max2(TxtW(t1, TH_SLP), TxtW(t2, TH_SLP)) * SC + 1 * SC
        If (cb - rb) - (ca + ra) > wl Then
            xm = ((ca + ra) + (cb - rb)) / 2

            PL_Text t2, xm, yM - r * 0.5, TH_SLP, "PF_SLOPE_T"
        Else
            ' ª�� ����: ���� ���⼱���� ǥ ����

Sub DrawBand(ByVal dS As Double, ByVal dE As Double, ByVal kind As Long, ByVal y As Double)
    Dim i As Long, f As Long, lst As Long, a As Long, kk As String, lt As String, dn As String
    Dim xA As Double, xB As Double, xc As Double, w As Double, tH As Double

    Dim lyL As String, lyM As String, lyT As String

    End If
    f = -1: lst = -1
    For i = 0 To pN - 1
        If pD(i) >= dS - 0.0001 And pD(i) <= dE + 0.0001 Then
            If f < 0 Then f = i
            lst = i
        End If
    Next
    If f < 0 Or lst <= f Then Exit Sub
    tH = TH_SEC * SC: upR = -1E+30: dnR = -1E+30
    PL_Line XD(pD(f), dS), y, XD(pD(lst), dS), y, lyL

    For i = f + 1 To lst

                    If p > 1 And kind = 1 Then
                        n1 = Trim$(Left$(dn, p - 1)): n2 = Mid$(dn, p)
                    Else
                        n1 = dn: n2 = ""
                    End If
                    w = Max2(Max2(TxtW(n1, TH_SEC), TxtW(n2, TH_SEC)), TxtW(lt, TH_SEC)) * SC
                    If xc - w / 2 > upR Then
                        sd2 = 1
                    ElseIf xc - w / 2 > dnR Then
                        sd2 = -1
                    Else
                        sd2 = 1: xc = upR + w / 2 + 1 * SC
                    End If
                    If sd2 = 1 Then
                        PL_Text lt, xc, y + 0.9 * tH, TH_SEC, lyT
                        If n2 <> "" Then
                            PL_Text n2, xc, y + 2.3 * tH, TH_SEC, lyT
                            PL_Text n1, xc, y + 3.7 * tH, TH_SEC, lyT
                        Else
                            PL_Text n1, xc, y + 2.3 * tH, TH_SEC, lyT
                        End If
                        upR = xc + w / 2
                    Else
                        PL_Text n1, xc, y - 0.9 * tH, TH_SEC, lyT

                        Else
                            PL_Text lt, xc, y - 2.3 * tH, TH_SEC, lyT
                        End If

                    End If

            End If
            a = i
        End If
    Next

Function BuildFacSeg() As Boolean
    Dim i As Long, c As String, vi() As Long, vC() As String, nV As Long, q As Long, lb() As String
    ReDim vi(pN), vC(pN), pFSeg(pN - 1)
    For i = 0 To pN - 1
        c = Left$(pFac(i), 1)
        If Len(c) > 0 Then
            If InStr(FACB_CODES, c) > 0 Then
                If nV > 0 Then
                    If Abs(pD(i) - pD(vi(nV - 1))) < 0.001 Then GoTo nextV
                End If
                vi(nV) = i: vC(nV) = c: nV = nV + 1
            End If
        End If
nextV:
    Next
    If nV = 0 Then Exit Function
    ReDim lb(nV)
    For q = 1 To nV - 1
        If FACB_LBL_START Then lb(q) = vC(q - 1) Else lb(q) = vC(q)
    Next
    q = 0
    For i = 0 To pN - 1
        If q < nV Then
            If i = vi(q) Then q = q + 1
        End If
        pFSeg(i) = q & "|" & lb(q)
    Next
    BuildFacSeg = True
End Function

Sub DrawNotes(ByVal dS As Double, ByVal dE As Double, ByVal xE As Double)
    Dim cnt As Long, i As Long, j As Long, k As Long, a As Double

    a = NOTE_TILT * PI / 180#
    For i = 0 To pN - 1
        If pD(i) >= dS - 0.0001 And pD(i) <= dE + 0.0001 Then
            f = FacName(pFac(i))
            If Len(f) > 0 Then
                DrawFacility pFac(i), XD(pD(i), dS), YE(pInv(i) + pDia(i) / 2000#)
                AddNote qX, qG, qW, qN, qT, qS, cnt, XD(pD(i), dS), pGL(i), f, "", StaText(pD(i))
            End If
            If Len(pTx1(i)) > 0 Then
                FitParts pTx1(i), pDia(i), nm, det
                AddNote qX, qG, qW, qN, qT, qS, cnt, XD(pD(i), dS), pGL(i), nm, det, StaText(pD(i))
            End If
        End If
    Next
    i = 0
    Do While i < cnt
        j = i: ext = qX(i) + qW(i) * Cos(a): gMax = qG(i)
        Do While j + 1 < cnt
            If qX(j + 1) > ext Then Exit Do
            j = j + 1

        Loop

        Next
        i = j + 1
    Loop
End Sub

' ���� ��ü���� ������� ��ġ ã�� (���� ǥ�����Ÿ�, ���� ���� ����)
Sub VB_Detect()
    Dim i As Long, m As Long, ok As Boolean
    Dim vd() As Double, vi() As Double, vk() As Long
    Dim vp() As Boolean, vu() As Boolean

    gVbN = 0: gVcN = 0
    If pN < 3 Then Exit Sub

    ReDim vd(0 To pN), vi(0 To pN), vk(0 To pN), vp(0 To pN), vu(0 To pN)

    m = 0
    For i = 0 To pN - 1
        ok = (pDia(i) > 0 And pInv(i) > 0)
        If Not ok Then
            VB_Run vd, vi, vk, vp, vu, m          ' ���� ���� �� ������� ���
            m = 0
        Else
            If m > 0 Then
                If Abs(pD(i) - vd(m - 1)) < VB_SAME_D Then
                    If Abs(pInv(i) - vi(m - 1)) < VB_STEP_TOL Then
                        ' ���� ���� �ߺ� ��: �ϳ��� ġ�� Pǥ�á�G�� �Է¸� ��ħ
                        If VB_IsP(i) Then vp(m - 1) = True
                        If InStr(CStr(pTx1(i)), "����") > 0 Then vu(m - 1) = True
                        GoTo nextV
                    End If
                    If Not VB_STEP_BEND Then      ' ���� �� ���� ����
                        VB_Run vd, vi, vk, vp, vu, m
                        m = 0
                    End If
                End If
            End If
            vd(m) = pD(i): vi(m) = pInv(i): vk(m) = i
            vp(m) = VB_IsP(i)
            vu(m) = (InStr(CStr(pTx1(i)), "����") > 0)
            m = m + 1
        End If
nextV:
    Next i
    VB_Run vd, vi, vk, vp, vu, m
End Sub

' �̾��� ���� �� ����: ������ ������ �� ���Ӱ� ���
Sub VB_Run(vd() As Double, vi() As Double, vk() As Long, vp() As Boolean, vu() As Boolean, ByVal m As Long)
    Dim keep() As Boolean, j As Long, nP As Long, a As Long, b As Long
    If m < 3 Then Exit Sub
    ReDim keep(0 To m - 1)
    keep(0) = True: keep(m - 1) = True

    nP = 0
    If VB_MODE = 1 Then
        For j = 1 To m - 2
            If vp(j) Then keep(j) = True: nP = nP + 1
        Next j
    End If
    If nP = 0 Then VB_DP vd, vi, keep, 0, m - 1   ' P�� ���� �� �ݿø� ���� ���� �� ���� ����

    a = 0: b = -1
    For j = 1 To m - 1
        If keep(j) Then
            If b < 0 Then
                b = j
            Else
                VB_Add vd, vi, vk, vu, a, b, j
                a = b: b = j
            End If
        End If
    Next j
End Sub

' ������ b���� ���� ��� �� ��Ͽ� �߰�
Sub VB_Add(vd() As Double, vi() As Double, vk() As Long, vu() As Boolean, _
           ByVal a As Long, ByVal b As Long, ByVal c As Long)
    Dim dx1 As Double, dy1 As Double, dx2 As Double, dy2 As Double, ang As Double
    dx1 = vd(b) - vd(a): dy1 = vi(b) - vi(a)
    dx2 = vd(c) - vd(b): dy2 = vi(c) - vi(b)
    ang = Abs(VB_Atan2(dy2, dx2) - VB_Atan2(dy1, dx1)) * 180# / PI
    If ang > 180# Then ang = 360# - ang

    If vu(b) Then                                   ' G�� ���� �Է� �켱
        VB_AddRef vd(b), ang, pIdx(vk(b)), "G�� ���� �Է�(�ڵ� ǥ�� �� ��)"
        Exit Sub
    End If
    If ang < VB_MIN_ANG Then
        If ang >= VB_CAND_MIN Then VB_AddRef vd(b), ang, pIdx(vk(b)), "�ּҰ��� �̸� ����"
        Exit Sub
    End If

    gVbD(gVbN) = vd(b)
    gVbAng(gVbN) = ang
    gVbDia(gVbN) = pDia(vk(b))
    gVbGL(gVbN) = pGL(vk(b))
    gVbRow(gVbN) = pIdx(vk(b))
    gVbS1(gVbN) = VB_Pct(dy1, dx1)
    gVbS2(gVbN) = VB_Pct(dy2, dx2)
    gVbN = gVbN + 1
End Sub

' �ܼ�ȭ(���۷���-��Ŀ): �������� VB_DP_TOL �Ѱ� ��� ���� ���������� ����
Sub VB_DP(vd() As Double, vi() As Double, keep() As Boolean, ByVal a As Long, ByVal b As Long)
    Dim j As Long, jm As Long, dmax As Double, dd As Double, t As Double, yl As Double
    If b - a < 2 Then Exit Sub
    dmax = -1#: jm = -1
    For j = a + 1 To b - 1
        If vd(b) - vd(a) > 0.000001 Then
            t = (vd(j) - vd(a)) / (vd(b) - vd(a))
            yl = vi(a) + t * (vi(b) - vi(a))
            dd = Abs(vi(j) - yl)
        Else
            dd = Abs(vi(j) - vi(a))
        End If
        If dd > dmax Then dmax = dd: jm = j
    Next j
    If dmax > VB_DP_TOL And jm > 0 Then
        keep(jm) = True
        VB_DP vd, vi, keep, a, jm
        VB_DP vd, vi, keep, jm, b
    End If
End Sub

Function VB_Atan2(ByVal dy As Double, ByVal dx As Double) As Double
    If Abs(dx) < 0.000001 Then
        If dy >= 0 Then VB_Atan2 = PI / 2# Else VB_Atan2 = -PI / 2#
    ElseIf dx > 0 Then
        VB_Atan2 = Atn(dy / dx)
    ElseIf dy >= 0 Then
        VB_Atan2 = Atn(dy / dx) + PI
    Else
        VB_Atan2 = Atn(dy / dx) - PI
    End If
End Function

Function BT_Draw(ByVal ttl As String, hdr As Variant, wd As Variant, cel() As String, rk() As Integer, _

Sub BT_AddItem(kd() As String, dd() As Double, di() As Double, sp() As String, ea() As Long, n As Long, _
               ByVal kn As String, ByVal d As Double, ByVal dia As Double, ByVal bSpec As String, ByVal cnt As Long)

Sub BT_Order(dd() As Double, ByVal n As Long, ord() As Long)
    Dim i As Long, j As Long, t As Long
    For i = 0 To n - 1
        ord(i) = i
    Next i
    For i = 1 To n - 1
        t = ord(i): j = i - 1
        Do While j >= 0
            If dd(ord(j)) <= dd(t) + 0.000001 Then Exit Do
            ord(j + 1) = ord(j)
            j = j - 1
        Loop
        ord(j + 1) = t
    Next i
End Sub

Function BT_First(kd() As String, di() As Double, sp() As String, ord() As Long, ByVal r As Long) As Boolean
    Dim q As Long, a As Long, k As Long
    k = ord(r)
    BT_First = True
    For q = 0 To r - 1
        a = ord(q)
        If kd(a) = kd(k) And sp(a) = sp(k) And Abs(di(a) - di(k)) < 0.5 Then
            BT_First = False
            Exit Function
        End If
    Next q
End Function

Function BT_Sum(kd() As String, di() As Double, sp() As String, ea() As Long, ByVal n As Long, ByVal k As Long) As Long
    Dim j As Long, t As Long

        If kd(j) = kd(k) And sp(j) = sp(k) And Abs(di(j) - di(k)) < 0.5 Then t = t + ea(j)
    Next j
    BT_Sum = t
End Function

Sub VLV_PostSheet(ByVal dS As Double, ByVal dE As Double)

    Dim vlX() As Double, vlY() As Double, vlC() As String, vlD() As Double

    If pN < 1 Then Exit Sub

    ReDim vlX(0 To pN), vlY(0 To pN), vlC(0 To pN), vlD(0 To pN)

        For i = 0 To pN - 1
            If pD(i) < dS - 0.001 Or pD(i) > dE + 0.001 Then GoTo nextI

            If pInv(i) <= 0 Then GoTo nextI

    VLV_Scan vlX, n
    If n > 0 Then
        VLV_Layer

            VLV_Insert vlC(j), vlX(j), vlY(j)

    End If

Sub VLV_Scan(vlX() As Double, ByVal n As Long)
    Dim i As Long, j As Long, o As Object, lay As String, s As String, s2 As String
    Dim pt As Variant, bn As String, hit As Boolean
    For i = ms.Count - 1 To gScan0 Step -1
        Set o = Nothing
        lay = ""
        On Error Resume Next
        Set o = ms.Item(i)
        lay = o.Layer
        On Error GoTo 0
        If o Is Nothing Then GoTo nextO

        If HB_RULE And lay = "PF_NOTE_T" Then
            s = ""
            On Error Resume Next
            s = o.TextString
            On Error GoTo 0
            s2 = HB_Fix(s)
            If s2 <> s Then o.TextString = s2
        ElseIf VLV_DEL_OLD And n > 0 And lay = "PF_FAC" Then
            pt = Empty: bn = ""
            On Error Resume Next
            pt = o.Center
            If IsEmpty(pt) Then pt = o.InsertionPoint
            bn = UCase(o.Name)
            On Error GoTo 0
            If Not IsEmpty(pt) And Left(bn, 2) <> "MH" Then
                hit = False
                For j = 0 To n - 1
                    If Abs(pt(0) - vlX(j)) < 0.01 Then hit = True: Exit For
                Next j
                If hit Then o.Delete
            End If
        End If
nextO:
    Next i
End Sub

Function HB_Fix(ByVal s As String) As String
    Dim p As Long, num As Double
    HB_Fix = s
    If UCase(Left(s, 2)) <> "D=" Then Exit Function
    If InStr(UCase(s), "EA") > 0 Then Exit Function      ' �̹� �ٲ� ����
    p = InStr(s, "��")
    If p = 0 Then p = InStr(3, LCase(s), "x")
    If p = 0 Then Exit Function
    num = Val(Mid(s, p + 1))
    If num <= 0 Then Exit Function
    If num = 11 Then num = 11.25
    If num = 22 Then num = 22.5
    HB_Fix = Left(s, p) & VB_Spec(num)
End Function

Sub VN_PostSheet(ByVal dS As Double, ByVal dE As Double)
    If Not VN_ON Then Exit Sub
    Dim vMs As Object, ent As Object, hObj As Object
    Dim nT As Long, i As Long, j As Long, k As Long, jBest As Long
    Dim tObj() As Object, tStr() As String, tX() As Double, tY() As Double
    Dim tR() As Double, tH() As Double, used() As Boolean
    Dim xA As Double, xB As Double, mg As Double, tmp As Double
    Dim s As String, sK As String, oNm As String, fCode As String
    Dim ip As Variant
    Dim cR As Double, sR As Double, dx As Double, dy As Double
    Dim pp As Double, aa As Double, ppBest As Double, gap As Double
    Dim dSta As Double, dep As Double, hasH As Boolean
    Dim p0(0 To 2) As Double, p1(0 To 2) As Double

    Set vMs = VN_MS()
    If vMs Is Nothing Then Exit Sub
    xA = XD(dS, dS): xB = XD(dE, dS)
    If xB < xA Then tmp = xA: xA = xB: xB = tmp
    mg = (xB - xA) * 0.15

    nT = 0
    ReDim tObj(0 To 99): ReDim tStr(0 To 99): ReDim tX(0 To 99)
    ReDim tY(0 To 99): ReDim tR(0 To 99): ReDim tH(0 To 99)

    On Error Resume Next
    For Each ent In vMs
        Err.Clear
        oNm = "": oNm = ent.ObjectName
        If InStr(1, oNm, "Text", vbTextCompare) > 0 And InStr(1, oNm, "MText", vbTextCompare) = 0 Then
            s = "": s = ent.TextString
            sK = UCase(Replace(s, " ", ""))
            If Left(sK, 3) = "NO." Or Left(sK, 2) = "H=" Or VN_CodeOf(sK) <> "" Then
                Err.Clear
                ip = ent.InsertionPoint
                If Err.Number = 0 Then
                    If ip(0) >= xA - mg And ip(0) <= xB + mg Then
                        If nT > UBound(tStr) Then
                            ReDim Preserve tObj(0 To nT + 99): ReDim Preserve tStr(0 To nT + 99)
                            ReDim Preserve tX(0 To nT + 99): ReDim Preserve tY(0 To nT + 99)
                            ReDim Preserve tR(0 To nT + 99): ReDim Preserve tH(0 To nT + 99)
                        End If
                        Set tObj(nT) = ent
                        tStr(nT) = sK: tX(nT) = ip(0): tY(nT) = ip(1)
                        tR(nT) = ent.Rotation: tH(nT) = ent.Height
                        nT = nT + 1
                    End If
                End If
            End If
        End If
    Next ent
    If nT = 0 Then Exit Sub
    ReDim used(0 To nT - 1)

    For i = 0 To nT - 1
        fCode = VN_CodeOf(tStr(i))
        If fCode <> "" Then
            cR = Cos(tR(i)): sR = Sin(tR(i))
            jBest = -1: ppBest = 1E+30
            ' ��� �̸� �ٷ� �Ʒ� ���� ���� ã��
            For j = 0 To nT - 1
                If j <> i And Not used(j) And Left(tStr(j), 3) = "NO." Then
                    If Abs(Sin(tR(j) - tR(i))) < 0.03 And Cos(tR(j) - tR(i)) > 0 Then
                        dx = tX(j) - tX(i): dy = tY(j) - tY(i)
                        pp = dx * sR - dy * cR          ' ���� �Ʒ��� �Ÿ�
                        aa = dx * cR + dy * sR          ' ���� ���� �Ÿ�
                        If pp > 0.2 * tH(i) And pp < 5 * tH(i) And Abs(aa) < 15 * tH(i) Then
                            If pp < ppBest Then ppBest = pp: jBest = j
                        End If
                    End If
                End If
            Next j

            If jBest >= 0 Then
                ' �̹� H= �� ������ �ǳʶ�(�ߺ� ����)
                hasH = False
                For k = 0 To nT - 1
                    If Left(tStr(k), 2) = "H=" Then
                        dx = tX(k) - tX(i): dy = tY(k) - tY(i)
                        pp = dx * sR - dy * cR: aa = dx * cR + dy * sR
                        If pp > 0 And pp < ppBest And Abs(aa) < 15 * tH(i) Then hasH = True: Exit For
                    End If
                Next k

                If Not hasH Then
                    dSta = VN_StaToD(tStr(jBest))
                    dep = VN_Depth(dSta, fCode)
                    If dSta >= 0 And dep > -900 Then
                        Err.Clear
                        Set hObj = Nothing
                        Set hObj = tObj(jBest).Copy
                        If Not hObj Is Nothing Then
                            hObj.TextString = "H=" & Format(dep, VN_FMT) & "m"
                            gap = VN_LGAP * tH(jBest)
                            p0(0) = 0: p0(1) = 0: p0(2) = 0
                            p1(0) = gap * Sin(tR(jBest))
                            p1(1) = -gap * Cos(tR(jBest))
                            p1(2) = 0
                            tObj(jBest).Move p0, p1      ' ���� ���ڸ� �� �� �Ʒ���
                        End If
                    End If
                End If
                used(jBest) = True
            End If
        End If
    Next i
    On Error GoTo 0
End Sub

Function VN_CodeOf(ByVal sK As String) As String
    Dim nm As Variant, cd As Variant, i As Long
    nm = Split(VN_NAMES, ","): cd = Split(VN_CODES, ",")
    For i = 0 To UBound(nm)
        If i > UBound(cd) Then Exit For
        If sK = UCase(Replace(Trim(nm(i)), " ", "")) Then
            VN_CodeOf = UCase(Trim(cd(i)))
            Exit Function
        End If
    Next i
End Function

Function VN_StaToD(ByVal sK As String) As Double
    Dim a As String, pos As Long
    VN_StaToD = -1
    If Left(sK, 3) <> "NO." Then Exit Function
    a = Mid(sK, 4)
    pos = InStr(a, "+")
    If pos > 0 Then
        VN_StaToD = Val(Left(a, pos - 1)) * VN_Chain() + Val(Mid(a, pos + 1))
    Else
        VN_StaToD = Val(a) * VN_Chain()
    End If
End Function

Function VN_Chain() As Double
    Static cVal As Double
    Dim ws As Object, v As Variant
    If cVal > 0 Then VN_Chain = cVal: Exit Function
    cVal = 20#
    On Error Resume Next
    Set ws = ActiveSheet
    If Not ws Is Nothing Then
        If Left(ws.Name, 3) = "����_" Then Set ws = Nothing
    End If
    If ws Is Nothing Then
        For Each ws In ActiveWorkbook.Worksheets
            If Left(ws.Name, 3) <> "����_" Then
                If IsNumeric(ws.Range("B2").Value) And IsNumeric(ws.Range("D2").Value) _
                   And Not IsEmpty(ws.Range("D2").Value) Then Exit For
            End If
        Next ws
    End If
    If Not ws Is Nothing Then
        v = ws.Range("D2").Value
        If IsNumeric(v) And Not IsEmpty(v) Then
            If CDbl(v) > 0 Then cVal = CDbl(v)
        End If
    End If
    VN_Chain = cVal
End Function

Function VN_Depth(ByVal d As Double, ByVal fCode As String) As Double
    Dim i As Long, i0 As Long, i1 As Long, f As String
    Dim h As Double, best As Double, hit As Boolean, t As Double
    VN_Depth = -999
    If d < 0 Then Exit Function
    i0 = LBound(pD): i1 = UBound(pD)
    If pN > 0 And i0 + pN - 1 < i1 Then i1 = i0 + pN - 1

    ' 1) ���� ���� + ���� �ü��ڵ� ��
    For i = i0 To i1
        If Abs(pD(i) - d) < 0.005 Then
            f = UCase(Trim(CStr(pFac(i))))
            If InStr(f, fCode) > 0 Then
                VN_Depth = pGL(i) - pInv(i)
                Exit Function
            End If
        End If
    Next i
    ' 2) ���� ���� �ٸ� �� (����=���� ����, �� ��=���� ���)
    hit = False
    For i = i0 To i1
        If Abs(pD(i) - d) < 0.005 Then
            h = pGL(i) - pInv(i)
            If Not hit Then
                best = h: hit = True
            ElseIf fCode = "I" Then
                If h > best Then best = h
            Else
                If h < best Then best = h
            End If
        End If
    Next i
    If hit Then VN_Depth = best: Exit Function
    ' 3) �յ� ����
    For i = i0 To i1 - 1
        If pD(i) <= d And pD(i + 1) >= d And pD(i + 1) > pD(i) Then
            t = (d - pD(i)) / (pD(i + 1) - pD(i))
            VN_Depth = (pGL(i) + (pGL(i + 1) - pGL(i)) * t) _
                     - (pInv(i) + (pInv(i + 1) - pInv(i)) * t)
            Exit Function
        End If
    Next i
End Function

Function VN_MS() As Object
    Dim app As Object, nm As Variant
    On Error Resume Next
    For Each nm In Array("ZWCAD.Application", "ZWCAD.Application.2025", _
                         "ZWCAD.Application.2024", "ZWCAD.Application.2023", "AutoCAD.Application")
        Set app = Nothing: Err.Clear
        Set app = GetObject(, CStr(nm))
        If Not app Is Nothing Then Exit For
    Next nm
    If app Is Nothing Then Exit Function
    Set VN_MS = app.ActiveDocument.ModelSpace
End Function

'   facOnly = True �̸� �ü��ڵ�(F��)�� ���� (��� ����ǥ��)

Sub ND_Hide(Optional ByVal facOnly As Boolean = False)
    Dim i As Long, j As Long, i0 As Long, i1 As Long
    Dim kF As String, kT As String
    gNdHid = False
    If Not ND_ON Then Exit Sub
    On Error GoTo EH
    i0 = LBound(pD): i1 = UBound(pD)
    If pN > 0 And i0 + pN - 1 < i1 Then i1 = i0 + pN - 1
    If UBound(pFac) < i1 Then i1 = UBound(pFac)
    If UBound(pTx1) < i1 Then i1 = UBound(pTx1)

    ReDim gNdFac(i0 To i1): ReDim gNdTx1(i0 To i1)
    For i = i0 To i1
        gNdFac(i) = pFac(i): gNdTx1(i) = pTx1(i)
    Next i
    gNdHid = True

    For i = i0 + 1 To i1
        kF = ND_Key(gNdFac(i))
        If facOnly Then
            kT = ""
        Else
            kT = ND_Key(gNdTx1(i))
        End If
        If kF <> "" Or kT <> "" Then
            For j = i0 To i - 1
                If Abs(pD(j) - pD(i)) < ND_TOL Then
                    If kF <> "" Then
                        If ND_Key(gNdFac(j)) = kF Then pFac(i) = "": kF = ""
                    End If
                    If kT <> "" Then
                        If ND_Key(gNdTx1(j)) = kT Then pTx1(i) = "": kT = ""
                    End If
                    If kF = "" And kT = "" Then Exit For
                End If
            Next j
        End If
    Next i
    Exit Sub
EH:
    ND_Restore
End Sub


Sub ND_Restore()
    Dim i As Long
    If Not gNdHid Then Exit Sub
    On Error Resume Next
    For i = LBound(gNdFac) To UBound(gNdFac)
        pFac(i) = gNdFac(i)
        pTx1(i) = gNdTx1(i)
    Next i
    gNdHid = False
End Sub

Function ND_Key(ByVal v As Variant) As String
    Dim s As String
    If IsError(v) Or IsNull(v) Or IsEmpty(v) Then Exit Function
    s = UCase(CStr(v))
    s = Replace(s, " ", ""): s = Replace(s, vbTab, "")
    s = Replace(s, ChrW(12288), "")          ' ���� ����
    ND_Key = s
End Function

Sub UL_PostSheet(ByVal dS As Double, ByVal dE As Double)
    If Not UL_ON Then Exit Sub
    Dim uMs As Object, ent As Object, nObj As Object
    Dim nT As Long, nLn As Long, i As Long, j As Long, k As Long
    Dim tObj() As Object, tStr() As String, tX() As Double, tY() As Double
    Dim tH() As Double, lObj() As Object
    Dim xA As Double, xB As Double, mg As Double, tmp As Double
    Dim oNm As String, s As String, ip As Variant
    Dim iName As Long, bestDy As Double, dx As Double, dy As Double
    Dim bMin As Variant, bMax As Variant
    Dim nLft As Double, nRgt As Double, nBot As Double, iTop As Double
    Dim yU As Double, hh As Double, xLft As Double, xRgt As Double
    Dim sp As Variant, ep As Variant, kBest As Long, kEnd As Integer
    Dim dd As Double, dBest As Double, oX As Double, xAtt As Double
    Dim dup As Boolean
    Dim p1(0 To 2) As Double, p2(0 To 2) As Double, pa(0 To 2) As Double

    Set uMs = VN_MS()
    If uMs Is Nothing Then Exit Sub
    xA = XD(dS, dS): xB = XD(dE, dS)
    If xB < xA Then tmp = xA: xA = xB: xB = tmp
    mg = (xB - xA) * 0.15

    nT = 0: nLn = 0
    ReDim tObj(0 To 99): ReDim tStr(0 To 99): ReDim tX(0 To 99)
    ReDim tY(0 To 99): ReDim tH(0 To 99): ReDim lObj(0 To 99)

    On Error Resume Next
    ' --- 1) ���ڿ� �� ������ ---
    For Each ent In uMs
        Err.Clear
        oNm = "": oNm = UCase(ent.ObjectName)
        If InStr(oNm, "TEXT") > 0 And InStr(oNm, "MTEXT") = 0 Then
            ip = ent.InsertionPoint
            If Err.Number = 0 Then
                If ip(0) >= xA - mg And ip(0) <= xB + mg And Abs(Sin(ent.Rotation)) < 0.02 Then
                    If nT > UBound(tStr) Then
                        ReDim Preserve tObj(0 To nT + 99): ReDim Preserve tStr(0 To nT + 99)
                        ReDim Preserve tX(0 To nT + 99): ReDim Preserve tY(0 To nT + 99)
                        ReDim Preserve tH(0 To nT + 99)
                    End If
                    Set tObj(nT) = ent
                    tStr(nT) = UCase(Replace(ent.TextString, " ", ""))
                    tX(nT) = ip(0): tY(nT) = ip(1): tH(nT) = ent.Height
                    nT = nT + 1
                End If
            End If
        ElseIf Right(oNm, 6) = "DBLINE" Then
            s = "": s = UCase(ent.Layer)
            If Left(s, 7) = "PF_UTIL" Or ent.Color = 1 Then
                sp = ent.StartPoint
                If sp(0) >= xA - mg And sp(0) <= xB + mg Then
                    If nLn > UBound(lObj) Then ReDim Preserve lObj(0 To nLn + 99)
                    Set lObj(nLn) = ent
                    nLn = nLn + 1
                End If
            End If
        End If
    Next ent
    If nT = 0 Then Exit Sub

    ' --- 2) INV.EL ���ڸ��� �� ��(�� �̸�) ã�� ���� ---
    For i = 0 To nT - 1
        If InStr(tStr(i), UL_KEY) > 0 Then
            iName = -1: bestDy = 1E+30
            For j = 0 To nT - 1
                If j <> i And Left(tStr(j), 3) <> "NO." And InStr(tStr(j), UL_KEY) = 0 Then
                    dx = tX(j) - tX(i): dy = tY(j) - tY(i)
                    If dy > 0.3 * tH(i) And dy < 3 * tH(i) And Abs(dx) < 3 * tH(i) Then
                        If dy < bestDy Then bestDy = dy: iName = j
                    End If
                End If
            Next j

            If iName >= 0 Then
                hh = tH(iName)
                ' �� �̸� ���� ����
                Err.Clear
                tObj(iName).GetBoundingBox bMin, bMax
                If Err.Number = 0 Then
                    nLft = bMin(0): nRgt = bMax(0): nBot = bMin(1)
                Else
                    nLft = tX(iName): nBot = tY(iName)
                    nRgt = tX(iName) + Len(tStr(iName)) * hh * 0.85
                End If
                ' INV ���� ����
                Err.Clear
                tObj(i).GetBoundingBox bMin, bMax
                If Err.Number = 0 Then
                    iTop = bMax(1)
                Else
                    iTop = tY(i) + tH(i)
                End If
                If iTop < nBot Then
                    yU = (nBot + iTop) / 2
                Else
                    yU = nBot - 0.25 * hh
                End If
                xLft = nLft - UL_PRE * hh
                xRgt = nRgt + UL_EXT * hh

                ' �̹� ������ ������ �ǳʶ�
                dup = False
                For k = 0 To nLn - 1
                    sp = lObj(k).StartPoint: ep = lObj(k).EndPoint
                    If Abs(sp(1) - yU) < 0.2 * hh And Abs(ep(1) - yU) < 0.2 * hh Then
                        If Abs(sp(0) - ep(0)) > hh And _
                           ((sp(0) + ep(0)) / 2) > xLft And ((sp(0) + ep(0)) / 2) < xRgt Then
                            dup = True: Exit For
                        End If
                    End If
                Next k

                If Not dup Then
                    ' �� �̸� �����̿��� ������ ���⼱ ã��
                    kBest = -1: dBest = 4 * hh
                    For k = 0 To nLn - 1
                        sp = lObj(k).StartPoint: ep = lObj(k).EndPoint
                        If Abs(sp(1) - ep(1)) > 0.2 * hh Then          ' ���μ� ����
                            dd = Sqr((sp(0) - nLft) ^ 2 + (sp(1) - yU) ^ 2)
                            If dd < dBest Then dBest = dd: kBest = k: kEnd = 0
                            dd = Sqr((ep(0) - nLft) ^ 2 + (ep(1) - yU) ^ 2)
                            If dd < dBest Then dBest = dd: kBest = k: kEnd = 1
                            dd = Sqr((sp(0) - nRgt) ^ 2 + (sp(1) - yU) ^ 2)
                            If dd < dBest Then dBest = dd: kBest = k: kEnd = 0
                            dd = Sqr((ep(0) - nRgt) ^ 2 + (ep(1) - yU) ^ 2)
                            If dd < dBest Then dBest = dd: kBest = k: kEnd = 1
                        End If
                    Next k

                    p1(0) = xLft: p1(1) = yU: p1(2) = 0
                    p2(0) = xRgt: p2(1) = yU: p2(2) = 0

                    If kBest >= 0 Then
                        ' ���⼱ ���� ���� ��(���幰 ��)���� �ű�
                        If UL_MOVE_LEADER Then
                            sp = lObj(kBest).StartPoint: ep = lObj(kBest).EndPoint
                            If kEnd = 0 Then oX = ep(0) Else oX = sp(0)
                            If oX > (nLft + nRgt) / 2 Then xAtt = xRgt Else xAtt = xLft
                            pa(0) = xAtt: pa(1) = yU: pa(2) = 0
                            If kEnd = 0 Then
                                lObj(kBest).StartPoint = pa
                            Else
                                lObj(kBest).EndPoint = pa
                            End If
                        End If
                        ' ���⼱�� ������ ���ٷ� (���̾�� ����)
                        Err.Clear
                        Set nObj = Nothing
                        Set nObj = lObj(kBest).Copy
                        If Not nObj Is Nothing Then
                            nObj.StartPoint = p1
                            nObj.EndPoint = p2
                        End If
                    Else
                        Err.Clear
                        Set nObj = Nothing
                        Set nObj = uMs.AddLine(p1, p2)
                        If Not nObj Is Nothing Then
                            Err.Clear
                            nObj.Layer = UL_LAYER
                            If Err.Number <> 0 Then nObj.Color = 1
                        End If
                    End If
                End If
            End If
        End If
    Next i
    On Error GoTo 0
End Sub
